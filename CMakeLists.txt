cmake_minimum_required(VERSION 3.24)
project(CodexBeerWidget VERSION 0.3.0 LANGUAGES CXX RC)
if(NOT WIN32 OR NOT MSVC)
  message(FATAL_ERROR "Windows with MSVC and Windows SDK is required")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.12.0/json.tar.xz
  URL_HASH SHA256=42f6e95cad6ec532fd372391373363b62a14af6d771056dbfc86160e6dfff7aa)
FetchContent_MakeAvailable(json)
add_library(widget_source src/codex_source.cpp src/settings.cpp src/worker.cpp src/sync_publisher.cpp src/sync_transport.cpp)
target_include_directories(widget_source PUBLIC src)
target_link_libraries(widget_source PUBLIC nlohmann_json::nlohmann_json psapi advapi32 winhttp crypt32)
target_compile_definitions(widget_source PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
target_compile_options(widget_source PUBLIC /W4 /permissive- /utf-8 /EHsc)
add_executable(CodexQuotaProbe src/probe_main.cpp src/probe.rc)
target_link_libraries(CodexQuotaProbe PRIVATE widget_source)
enable_testing()
add_executable(protocol_tests tests/protocol_tests.cpp src/tests.rc)
target_link_libraries(protocol_tests PRIVATE widget_source)
add_test(NAME protocol COMMAND protocol_tests)
set_tests_properties(protocol PROPERTIES TIMEOUT 45)
add_executable(model_tests tests/model_tests.cpp src/tests.rc)
target_link_libraries(model_tests PRIVATE widget_source)
add_test(NAME model COMMAND model_tests)
add_executable(sync_snapshot_tests tests/sync_snapshot_tests.cpp src/tests.rc)
target_link_libraries(sync_snapshot_tests PRIVATE widget_source)
target_compile_definitions(sync_snapshot_tests PRIVATE SYNC_FIXTURES="${CMAKE_CURRENT_SOURCE_DIR}/protocol/fixtures/normalization.json")
add_test(NAME sync_snapshot COMMAND sync_snapshot_tests)
add_executable(sync_publisher_tests tests/sync_publisher_tests.cpp src/tests.rc)
target_link_libraries(sync_publisher_tests PRIVATE widget_source)
add_test(NAME sync_publisher COMMAND sync_publisher_tests)
set_tests_properties(sync_publisher PROPERTIES TIMEOUT 15)
add_executable(CodexBeerWidget WIN32 src/app.cpp src/renderer.cpp src/settings_ui.cpp src/app.manifest src/app.rc)
target_link_libraries(CodexBeerWidget PRIVATE widget_source d2d1 d3d11 dxgi dcomp dwrite user32 gdi32 ole32 shell32 comctl32 comdlg32 wtsapi32 uuid)
target_link_options(CodexBeerWidget PRIVATE /MANIFEST:EMBED /MANIFESTUAC:"level='asInvoker' uiAccess='false'")
target_link_libraries(CodexBeerWidget PRIVATE delayimp)
target_link_options(CodexBeerWidget PRIVATE /DELAYLOAD:d3d11.dll /DELAYLOAD:dcomp.dll /DELAYLOAD:d2d1.dll /DELAYLOAD:DWrite.dll)

install(TARGETS CodexBeerWidget RUNTIME DESTINATION . COMPONENT Runtime)
install(FILES README.md NOTICE.txt DESTINATION . COMPONENT Runtime)
install(DIRECTORY docs/ DESTINATION docs COMPONENT Runtime)
install(FILES "${json_SOURCE_DIR}/LICENSE.MIT" DESTINATION licenses RENAME nlohmann-json-MIT.txt COMPONENT Runtime)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_NAME CodexBeerWidget)
set(CPACK_PACKAGE_VERSION ${PROJECT_VERSION})
set(CPACK_PACKAGE_FILE_NAME "CodexBeerWidget-${PROJECT_VERSION}-windows-x64")
set(CPACK_COMPONENTS_ALL Runtime)
set(CPACK_ARCHIVE_COMPONENT_INSTALL ON)
set(CPACK_COMPONENTS_GROUPING ALL_COMPONENTS_IN_ONE)
include(CPack)
